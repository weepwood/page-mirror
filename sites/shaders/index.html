GlassBackground.tsx
import { ... } from 'shaders/react'

export default function GlassBackground() {
  return (
    <Shader>
      {/* Ambient background */}
      <Swirl
        blend={50}
        detail={1.7}
        stops={["#e6e2ff", "#bdb2ff"]} />
      {/* Interactive cursor effect */}
      <ChromaFlow
        baseColor="#ae00ff"
        downColor="#4642ff"
        leftColor="#5900ff"
        rightColor="#5b4fff"
        upColor="#7105ff" />
      {/* Glass-like distortion */}
      <FlutedGlass
        aberration={0.25}
        angle={30}
        frequency={8}
        highlightColor="#ffffff"
        refraction={4}
        softness={0.5}/>
    </Shader>
  )
}
